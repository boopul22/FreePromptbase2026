# robots.txt — groups are NOT additive: a named user-agent obeys ONLY its own
# group. So the Disallow set must be repeated per named bot, otherwise "welcoming"
# a bot would accidentally un-block /admin, /api, etc. for that bot.

Disallow: /admin/
Disallow: /api/
Disallow: /account
Disallow: /saved
